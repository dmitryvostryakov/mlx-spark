cmake_minimum_required(VERSION 3.25)
project(mlx_spark_scaffold LANGUAGES CXX VERSION 0.0.0)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(MLX_SPARK_ENABLE_CUDA "Build opt-in, unvalidated CUDA probe/reference" OFF)
include(CTest)
add_library(mlx_spark_contracts STATIC native/src/contracts.cpp)
target_include_directories(mlx_spark_contracts PUBLIC native/include)
add_executable(test_contracts native/tests/test_contracts.cpp)
target_link_libraries(test_contracts PRIVATE mlx_spark_contracts)
add_test(NAME host_contracts COMMAND test_contracts)
if(MLX_SPARK_ENABLE_CUDA)
  enable_language(CUDA)
  find_package(CUDAToolkit REQUIRED)
  set(CMAKE_CUDA_STANDARD 17)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  add_executable(spark_device_probe native/cuda/device_probe.cu)
  target_link_libraries(spark_device_probe PRIVATE CUDA::cudart)
  add_library(spark_gdn_reference STATIC native/cuda/gdn_decode_reference.cu)
  target_include_directories(spark_gdn_reference PUBLIC native/include)
  target_link_libraries(spark_gdn_reference PUBLIC CUDA::cudart)
  add_executable(test_gdn_cuda native/tests/test_gdn_cuda.cu)
  target_link_libraries(test_gdn_cuda PRIVATE spark_gdn_reference)
  add_test(NAME unvalidated_cuda_gdn_reference COMMAND test_gdn_cuda)
  set_tests_properties(unvalidated_cuda_gdn_reference PROPERTIES SKIP_RETURN_CODE 77 LABELS "gpu;reference")
endif()
